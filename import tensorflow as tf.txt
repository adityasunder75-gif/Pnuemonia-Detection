import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout, BatchNormalization
from tensorflow.keras.preprocessing.image import ImageDataGenerator
import os
import matplotlib.pyplot as plt
import numpy as np

# --- 1. Configuration and Constants ---
# Define the size to which all images will be resized
IMAGE_SIZE = 150
# Define the batch size for training
BATCH_SIZE = 32
# Set the number of training epochs (can be increased for better results)
EPOCHS = 10

# IMPORTANT: Define the root path where your dataset is located.
# The folder structure should be:
# root_dir/
# ├── train/
# │   ├── NORMAL/
# │   └── PNEUMONIA/
# ├── val/
# │   ├── NORMAL/
# │   └── PNEUMONIA/
# └── test/
#     ├── NORMAL/
#     └── PNEUMONIA/
ROOT_DIR = './chest_xray' # <--- CHANGE THIS PATH to your dataset location

if not os.path.exists(ROOT_DIR):
    print("WARNING: The dataset path was not found. Please set the 'ROOT_DIR' variable to the correct path.")
    print("Using dummy directory names to allow the script structure to be viewed.")
    # Use dummy paths for structure visualization if the real path is missing
    train_dir = 'dummy_train'
    val_dir = 'dummy_val'
    test_dir = 'dummy_test'
else:
    train_dir = os.path.join(ROOT_DIR, 'train')
    val_dir = os.path.join(ROOT_DIR, 'val')
    test_dir = os.path.join(ROOT_DIR, 'test')


# --- 2. Data Preprocessing and Augmentation ---

# 2.1. Data Augmentation for Training Set
# Data augmentation helps prevent overfitting by artificially creating variations
# of the training images (e.g., rotation, zoom, shear). All images are rescaled
# by 1/255 to normalize pixel values from 0-255 to 0-1.
train_datagen = ImageDataGenerator(
    rescale=1./255,
    shear_range=0.2,
    zoom_range=0.2,
    horizontal_flip=True
)

# 2.2. Data Generation for Validation and Test Sets
# Validation and Test sets are only rescaled (normalized), but not augmented.
test_val_datagen = ImageDataGenerator(rescale=1./255)

# 2.3. Flow Data from Directories
# This creates generators that load batches of images directly from the folders.

# Training Generator
train_generator = train_datagen.flow_from_directory(
    train_dir,
    target_size=(IMAGE_SIZE, IMAGE_SIZE), # Resize all images to this size
    batch_size=BATCH_SIZE,
    class_mode='binary' # Binary classification (Pneumonia or Normal)
)

# Validation Generator
validation_generator = test_val_datagen.flow_from_directory(
    val_dir,
    target_size=(IMAGE_SIZE, IMAGE_SIZE),
    batch_size=BATCH_SIZE,
    class_mode='binary'
)

# Test Generator
test_generator = test_val_datagen.flow_from_directory(
    test_dir,
    target_size=(IMAGE_SIZE, IMAGE_SIZE),
    batch_size=BATCH_SIZE,
    class_mode='binary',
    shuffle=False # Crucial for later evaluation to match predictions with true labels
)

# --- 3. Model Definition (Convolutional Neural Network - CNN) ---
# A sequential model is built using multiple Convolutional and Pooling layers
# followed by Dense layers for classification. [Image of Convolutional Neural Network architecture]

model = Sequential([
    # First Convolutional Block
    Conv2D(32, (3, 3), activation='relu', input_shape=(IMAGE_SIZE, IMAGE_SIZE, 3)),
    BatchNormalization(),
    MaxPooling2D((2, 2)),
    
    # Second Convolutional Block
    Conv2D(64, (3, 3), activation='relu'),
    BatchNormalization(),
    MaxPooling2D((2, 2)),
    
    # Third Convolutional Block
    Conv2D(128, (3, 3), activation='relu'),
    BatchNormalization(),
    MaxPooling2D((2, 2)),
    
    # Flattening: Converts the 3D feature maps to 1D feature vectors
    Flatten(),
    
    # Fully Connected Layer (Dense)
    Dense(512, activation='relu'),
    Dropout(0.5), # Dropout layer helps reduce overfitting
    
    # Output Layer
    # Since this is binary classification (0 or 1), a single neuron with sigmoid activation is used.
    Dense(1, activation='sigmoid')
])

# --- 4. Model Compilation ---
model.compile(
    loss='binary_crossentropy', # Appropriate loss function for binary classification
    optimizer='adam',
    metrics=['accuracy']
)

# Display a summary of the model structure
model.summary()

# --- 5. Model Training ---
print("\n--- Starting Model Training ---")

# The model is trained using the data generators
history = model.fit(
    train_generator,
    steps_per_epoch=train_generator.samples // BATCH_SIZE, # Number of batches per epoch
    epochs=EPOCHS,
    validation_data=validation_generator,
    validation_steps=validation_generator.samples // BATCH_SIZE
)

# --- 6. Model Evaluation ---
print("\n--- Evaluating Model on Test Data ---")

# Evaluate the model on the unseen test set
loss, accuracy = model.evaluate(test_generator, steps=test_generator.samples // BATCH_SIZE)
print(f"Test Loss: {loss:.4f}")
print(f"Test Accuracy: {accuracy:.4f}")

# Generate predictions and confusion matrix
# Get the true labels from the test generator
Y_true = test_generator.classes
# Get the predicted probabilities
Y_pred_prob = model.predict(test_generator)
# Convert probabilities to binary class labels (0 or 1)
Y_pred_classes = (Y_pred_prob > 0.5).astype(int).flatten()

# Print Classification Report
from sklearn.metrics import classification_report, confusion_matrix
print("\nClassification Report:")
target_names = ['NORMAL', 'PNEUMONIA']
print(classification_report(Y_true[:len(Y_pred_classes)], Y_pred_classes, target_names=target_names))

# Optional: Plot training history
def plot_history(history):
    plt.figure(figsize=(12, 4))
    
    # Plot Loss
    plt.subplot(1, 2, 1)
    plt.plot(history.history['loss'], label='Training Loss')
    plt.plot(history.history['val_loss'], label='Validation Loss')
    plt.title('Loss Curves')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.legend()
    
    # Plot Accuracy
    plt.subplot(1, 2, 2)
    plt.plot(history.history['accuracy'], label='Training Accuracy')
    plt.plot(history.history['val_accuracy'], label='Validation Accuracy')
    plt.title('Accuracy Curves')
    plt.xlabel('Epoch')
    plt.ylabel('Accuracy')
    plt.legend()
    
    plt.show()

# --- 8. Example Prediction Function ---
def predict_new_image(model, img_path):
    """
    Loads a single image, preprocesses it, and makes a prediction.
    """
    try:
        from tensorflow.keras.preprocessing import image
        img = image.load_img(img_path, target_size=(IMAGE_SIZE, IMAGE_SIZE))
        img_array = image.img_to_array(img)
        # Expand dimensions to match model input shape (1, 150, 150, 3)
        img_array = np.expand_dims(img_array, axis=0)
        # Rescale the image like the training data
        img_array /= 255.0

        # Make prediction
        prediction = model.predict(img_array)[0][0]
        
        print(f"\nPrediction for {os.path.basename(img_path)}:")
        print(f"Pneumonia Probability: {prediction * 100:.2f}%")
        
        if prediction > 0.5:
            print("Diagnosis: PNEUMONIA (Predicted)")
        else:
            print("Diagnosis: NORMAL (Predicted)")
            
    except FileNotFoundError:
        print(f"\nERROR: Image file not found at path: {img_path}")
    except Exception as e:
        print(f"\nAn error occurred during prediction: {e}")
